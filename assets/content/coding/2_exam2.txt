    for (int *p = arr + 1; p < arr + size; p++) {

    return max;

solution:
void reverse_array(int *arr, int size) {
    int *left = arr;
    int *right = arr + size - 1;
    while (left < right) {
        int temp = *left;
        *left = *right;
        *right = temp;
        left++;
        right--;
    }
}

    }
    return count;
}

};
@end

}

void PrintList(Node* headPtr) {
    // your code here
}
@end
helpers:

    std::string s;
    int guard = 0;

+ InsertAtEnd builds 10 20 30 in order

InsertAtEnd(&head, 5);
EXPECT_TRUE(head != nullptr);
EXPECT_EQ(head->data, 5);
EXPECT_TRUE(head->nextPtr == nullptr);
- the last node's nextPtr is nullptr

while (p && p->nextPtr) p = p->nextPtr;
EXPECT_EQ(p->data, 4);
EXPECT_TRUE(p->nextPtr == nullptr);
- PrintList of an empty list prints no values

@end
hint: Node* newNode = new Node(); newNode->data = newData; newNode->nextPtr = nullptr; if (*ptrToHeadPtr == nullptr) { *ptrToHeadPtr = newNode; return; }  otherwise walk to the last node and link it.
solution:
void InsertAtEnd(Node** ptrToHeadPtr, int newData) {
    Node* newNode = new Node();
    newNode->data = newData;
    newNode->nextPtr = nullptr;
    if (*ptrToHeadPtr == nullptr) {
        *ptrToHeadPtr = newNode;
        return;
    }
    Node* cur = *ptrToHeadPtr;
    while (cur->nextPtr != nullptr) cur = cur->nextPtr;
    cur->nextPtr = newNode;
}

void PrintList(Node* headPtr) {
    Node* cur = headPtr;
    while (cur != nullptr) {
        cout << cur->data << " ";
        cur = cur->nextPtr;
    }
    cout << endl;
}
@end

@p e2.ll_delete
title: Linked List: DeleteWithValue
topic: linkedlist
diff: Medium
desc:
Delete the **first** node holding `value`, re-linking the list around it, and free it with `delete`.

- If the list is empty print `List is empty.` and return.

void DeleteWithValue(Node** ptrToHeadPtr, int value) {
    if (*ptrToHeadPtr == nullptr) {
        cout << "List is empty." << endl;
        return;
    }
    Node* cur = *ptrToHeadPtr;
    if (cur->data == value) {
        *ptrToHeadPtr = cur->nextPtr;
        delete cur;
        return;
    }
    Node* prev = nullptr;
    while (cur != nullptr && cur->data != value) {
        prev = cur;
        cur = cur->nextPtr;
    }
    if (cur == nullptr) {
        cout << "Value not found in the list." << endl;
        return;
    }
    prev->nextPtr = cur->nextPtr;
    delete cur;
}

@p e2.ll_class
title: LinkedList Class: Size, Search & Deep Copy
topic: classes
diff: Medium
desc:
The C++ **class** version of the list from the lecture. `insertAtEnd`, `clear`, the destructor and `toString` are written for you. Implement:

- `insertAtBeginning(int)`
- `getSize()` - number of nodes
- `contains(int)` - is the value in the list?
- the **copy constructor** - a *deep* copy: new nodes with the same values
- the **assignment operator** - clear this list, then deep-copy the other one (and survive `a = a;`)

Why does the copy matter? The default copy just copies the `head` pointer, so both lists share nodes - and both destructors delete them. That double delete crashes the program.
@end
support:
class Node {
public:
    int data;
    Node* next;
    Node(int value) { data = value; next = nullptr; }
};
@end
starter:

class LinkedList {

    Node* head;
public:
    LinkedList() { head = nullptr; }
    ~LinkedList() { clear(); }

        head = nullptr;
        for (Node* c = other.head; c != nullptr; c = c->next) insertAtEnd(c->data);
    }

    LinkedList& operator=(const LinkedList& other) {
        if (this != &other) {
            clear();
            for (Node* c = other.head; c != nullptr; c = c->next) insertAtEnd(c->data);
        }
        return *this;
    }

    void insertAtEnd(int newData) {
        Node* newNode = new Node(newData);

class DoublyLinkedNode {
public:
    int data;
    DoublyLinkedNode* prevPointer;
    DoublyLinkedNode* nextPointer;
    DoublyLinkedNode(int value) : data(value), prevPointer(nullptr), nextPointer(nullptr) {}
};

class DoublyLinkedList {
public:
    DoublyLinkedNode* head;
    DoublyLinkedNode* tail;

    DoublyLinkedList() : head(nullptr), tail(nullptr) {}

    void InsertAtBeginning(int value) {
        DoublyLinkedNode* n = new DoublyLinkedNode(value);
        n->nextPointer = head;
        if (head != nullptr) head->prevPointer = n;
        else tail = n;
        head = n;
    }

    void InsertAtEnd(int value) {
        DoublyLinkedNode* n = new DoublyLinkedNode(value);
        n->prevPointer = tail;
        if (tail != nullptr) tail->nextPointer = n;
        else head = n;
        tail = n;
    }

    void InsertAtPosition(int value, int position) {
        if (position <= 0 || head == nullptr) { InsertAtBeginning(value); return; }
        DoublyLinkedNode* cur = head;
        int index = 0;
        while (cur != nullptr && index < position) { cur = cur->nextPointer; index++; }
        if (cur == nullptr) { InsertAtEnd(value); return; }
        DoublyLinkedNode* n = new DoublyLinkedNode(value);
        n->prevPointer = cur->prevPointer;
        n->nextPointer = cur;
        cur->prevPointer->nextPointer = n;
        cur->prevPointer = n;
    }
};
