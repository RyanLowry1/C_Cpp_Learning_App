@end
hint: Inside a method you can read other.x directly - private means private to the class, not to the object.
solution:
class Point {
private:
    double x, y;
public:
    Point() : x(0), y(0) {}
    Point(double nx, double ny) : x(nx), y(ny) {}
    double getX() { return x; }
    double getY() { return y; }
    void offset(double dx, double dy) { x += dx; y += dy; }
    double distanceTo(const Point& other) {
        double dx = x - other.x, dy = y - other.y;
        return sqrt(dx * dx + dy * dy);
    }
};

class Box {
public:
    double getVolume(void) { return length * breadth * height; }
    void setLength(double len) { length = len; }
    void setBreadth(double bre) { breadth = bre; }
    void setHeight(double hei) { height = hei; }
    Box operator+(const Box& b) {
        Box box;
        box.length = this->length + b.length;
        box.breadth = this->breadth + b.breadth;
        box.height = this->height + b.height;
        return box;
    }
private:
    double length = 0;
    double breadth = 0;
    double height = 0;
};

@p cpp.clock
title: Clock: Prefix and Postfix ++
topic: operators
diff: Medium
desc:
From the notes on overloading increment. A `Clock` stores hours and minutes. `tick()` advances it one minute (wrapping 59 -> 00 and 23 -> 00).

- **Prefix** `++c` advances the clock and returns the **new** value.
- **Postfix** `c++` advances the clock but returns a copy of the **old** value. The dummy `int` parameter is how C++ tells them apart.
- `operator<<` prints the time as `HH:MM` with leading zeros.
@end
example:
Clock c1, c2(3,20), c3(10,59);
cout << c1 << " " << c2 << " " << c3;   // 12:00 03:20 10:59
c1++; ++c2; c3++;                         // 12:01 03:21 11:00
c1 = ++c2;                                // c1 03:22, c2 03:22
c1 = c3++;                                // c1 11:00, c3 11:01
@end
starter:
class Clock {
    friend ostream& operator<<(ostream& out, const Clock& c);
public:
    Clock(int h = 12, int m = 0) : hour(h), min(m) {}
    Clock operator++() {         // prefix
        // TODO
        return *this;
    }
    Clock operator++(int) {      // postfix
        // TODO
        return *this;
    }
private:
    void tick() {
        min = (min + 1) % 60;
        if (min == 0) hour = (hour + 1) % 24;
    }
    int hour, min;
};

ostream& operator<<(ostream& os, const Clock& c) {
    // TODO: print HH:MM
    return os;
}
@end
tests:
+ printing pads with zeros
Clock c1, c2(3, 20), c3(10, 59);
EXPECT_OUT(cout << c1 << " " << c2 << " " << c3, "12:00 03:20 10:59");
+ c1 = ++c2 gives both the new time
Clock c1, c2(3, 21);
c1 = ++c2;
EXPECT_OUT(cout << c1 << " " << c2, "03:22 03:22");
- c1 = c3++ gives c1 the old time
Clock c1, c3(11, 0);
c1 = c3++;
EXPECT_OUT(cout << c1 << " " << c3, "11:00 11:01");
- increments roll over the hour
Clock c(10, 59);
c++;
EXPECT_OUT(cout << c, "11:00");
- prefix and postfix both advance the clock
Clock a(1, 1), b(1, 1);
++a;
b++;
EXPECT_OUT(cout << a << " " << b, "01:02 01:02");
- midnight wraps to 00:00
Clock c(23, 59);
++c;
EXPECT_OUT(cout << c, "00:00");
@end
hint: Prefix: tick(); return *this;  Postfix: Clock old = *this; tick(); return old;  Printing: if (c.hour < 10) os << '0'; os << c.hour << ":"; ...
solution:
class Clock {
    friend ostream& operator<<(ostream& out, const Clock& c);
public:
    Clock(int h = 12, int m = 0) : hour(h), min(m) {}
    Clock operator++() {
        tick();
        return *this;
    }
    Clock operator++(int) {
        Clock old = *this;
        tick();
        return old;
    }
private:
    void tick() {
        min = (min + 1) % 60;
        if (min == 0) hour = (hour + 1) % 24;
    }
    int hour, min;
};

ostream& operator<<(ostream& os, const Clock& c) {
    if (c.hour < 10) os << '0';
    os << c.hour << ":";
    if (c.min < 10) os << '0';
    os << c.min;
    return os;
}
@end

class Student {
public:
    int studentID;
    char* name;

    Student(int id, const char* n) {
        studentID = id;
        name = new char[strlen(n) + 1];
        strcpy(name, n);
    }

    Student(const Student& o) {
        studentID = o.studentID;
        name = new char[strlen(o.name) + 1];
        strcpy(name, o.name);

        char* fresh = new char[strlen(n) + 1];
        strcpy(fresh, n);
        delete[] name;
        name = fresh;
    }

    ~Student() { delete[] name; }
};

    for (int i = 0; i < ELEMENTS; i++) total += pArray[i];

+ mismatched brackets

- an unclosed opener

- a closer with nothing open

- an empty string is balanced

- nesting in the wrong order

- deep nesting
EXPECT_TRUE(balanced("{[()()]}([])"));
@end
hint: For each char: if it opens, push it. If it closes: the stack must be non-empty and top() must be the matching opener - then pop(). At the end the stack must be empty.
solution:
bool balanced(const string& s) {
    stack<char> st;
    for (char c : s) {
        if (c == '(' || c == '[' || c == '{') st.push(c);
        else if (c == ')' || c == ']' || c == '}') {
            if (st.empty()) return false;
            char open = st.top();
            st.pop();
            if ((c == ')' && open != '(') || (c == ']' && open != '[') || (c == '}' && open != '{')) return false;
        }
    }
    return st.empty();
}
@end

int** make2D(int rows, int cols) {
    int** arr = new int*[rows];
    int counter = 1;
    for (int r = 0; r < rows; r++) {
        arr[r] = new int[cols];
        for (int c = 0; c < cols; c++) arr[r][c] = counter++;
    }
    return arr;
}

void free2D(int** arr, int rows) {
    for (int r = 0; r < rows; r++) delete[] arr[r];
    delete[] arr;
}
