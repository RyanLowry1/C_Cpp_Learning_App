    void tick() { if (++min == 60) { min = 0; hour = (hour + 1) % 24; } }
    Clock operator++() { tick(); return *this; }
    Clock operator++(int) { Clock old = *this; tick(); return old; }
    friend ostream& operator<<(ostream& out, const Clock& c) {
        out << c.hour << ":" << (c.min < 10 ? "0" : "") << c.min;
        return out;
    }
private:
    int hour, min;
};
int main() {
    Clock c(9, 59);
    Clock a = c++;
    Clock b = ++c;
    cout << a << " " << b << " " << c << endl;
}
@end
ans: 9:59 10:01 10:01
why: `c++` returns the old value (9:59) and ticks c to 10:00. `++c` ticks to 10:01 and returns the new value.
run: yes
