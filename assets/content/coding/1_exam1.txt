    return true;
}

int convertBinarytoDecimal(int binaryNumber) {
    int result = 0;
    int place = 1;
    while (binaryNumber > 0) {
        result += (binaryNumber % 10) * place;
        place *= 2;
        binaryNumber /= 10;
    }
    return result;
}

hint: Track two variables, largest and second, both starting at -1. For each value: if it beats largest, the old largest becomes second; else if it is between them (and not equal to largest), it becomes second.
solution:
int find2ndLargest(int size, int arr[]) {
    int largest = -1, second = -1;
    for (int i = 0; i < size; i++) {
        if (arr[i] > largest) {
            second = largest;
            largest = arr[i];
        } else if (arr[i] < largest && arr[i] > second) {
            second = arr[i];
        }
    }
    return second;
}
@end

@end

bool insertElement(int& size, int& count, int arr[], int elementToInsert, int insertIndex) {
    if (count >= size || insertIndex < 0 || insertIndex > count) return false;
    for (int i = count; i > insertIndex; i--) {
        arr[i] = arr[i - 1];
    }
    arr[insertIndex] = elementToInsert;
    count++;
    return true;
}

int ReplaceWithMinusOne(int array[4][6])
{
    for (int r = 0; r < 4; r++) {
        for (int c = 0; c < 6; c++) {
            int v = array[r][c];
            if (v < 1 || v > 4) continue;
            int len = 1;
            while (c + len < 6 && array[r][c + len] == v) len++;
            if (len >= 3) {
                for (int k = 0; k < len; k++) array[r][c + k] = -1;
                return v * len;
            }
            len = 1;
            while (r + len < 4 && array[r + len][c] == v) len++;
            if (len >= 3) {
                for (int k = 0; k < len; k++) array[r + k][c] = -1;
                return v * len;
            }
        }
    }
    return 0;
}

static bool isIdentChar(char c) {
    return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_';
}

void FindAllVariables(char variables[], char code[])
{
    int out = 0;
    variables[0] = '\0';
    int i = 0;
    while (code[i] != '\0') {
        // start of a line: skip indentation
        while (code[i] == ' ' || code[i] == '\t') i++;
        int skip = 0;
        if (strncmp(&code[i], "int", 3) == 0 && (code[i + 3] == ' ' || code[i + 3] == '\t')) skip = 3;
        else if (strncmp(&code[i], "char", 4) == 0 && (code[i + 4] == ' ' || code[i + 4] == '\t')) skip = 4;
        if (skip > 0) {
            int p = i + skip;
            while (code[p] == ' ' || code[p] == '\t') p++;
            int start = p;
            while (isIdentChar(code[p])) p++;
            int q = p;
            while (code[q] == ' ') q++;
            if (p > start && code[q] != '(') {
                if (out > 0) variables[out++] = ',';
                for (int k = start; k < p; k++) variables[out++] = code[k];
                variables[out] = '\0';
            }
        }
        // move to the next line
        while (code[i] != '\0' && code[i] != '\n') i++;
        if (code[i] == '\n') i++;
    }
}
@end

hint: Keep arrays of names and depths (a stack). For each line up to lineNumber: first check for a declaration and push (name, current depth); then scan the line's characters, doing depth++ on '{' and on '}' doing depth-- and popping every variable whose depth is now greater than the current depth.
solution:
static bool isIdentChar2(char c) {
    return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_';
}

void FindAllVariablesInScope(char variables[], char code[], int lineNumber)
{
    char names[50][32];
    int depths[50];
    int count = 0;
    int depth = 0;
    int line = 1;
    int i = 0;
    while (code[i] != '\0' && line <= lineNumber) {
        int lineStart = i;
        while (code[i] == ' ' || code[i] == '\t') i++;
        int skip = 0;
        if (strncmp(&code[i], "int", 3) == 0 && (code[i + 3] == ' ' || code[i + 3] == '\t')) skip = 3;
        else if (strncmp(&code[i], "char", 4) == 0 && (code[i + 4] == ' ' || code[i + 4] == '\t')) skip = 4;
        if (skip > 0) {
            int p = i + skip;
            while (code[p] == ' ' || code[p] == '\t') p++;
            int start = p;
            while (isIdentChar2(code[p])) p++;
            int q = p;
            while (code[q] == ' ') q++;
            if (p > start && code[q] != '(' && count < 50) {
                int len = p - start;
                if (len > 31) len = 31;
                strncpy(names[count], &code[start], len);
                names[count][len] = '\0';
                depths[count] = depth;
                count++;
            }
        }
        // braces on this line
        for (int k = lineStart; code[k] != '\0' && code[k] != '\n'; k++) {
            if (code[k] == '{') depth++;
            else if (code[k] == '}') {
                depth--;
                while (count > 0 && depths[count - 1] > depth) count--;
            }
        }
        while (code[i] != '\0' && code[i] != '\n') i++;
        if (code[i] == '\n') i++;
        line++;
    }
    variables[0] = '\0';
    for (int v = 0; v < count; v++) {
        if (v > 0) strcat(variables, ",");
        strcat(variables, names[v]);
    }
}
